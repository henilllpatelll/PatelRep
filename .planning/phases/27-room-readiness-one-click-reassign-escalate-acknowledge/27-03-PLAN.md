```tsx
import { useState } from 'react'
import { Button, IconButton } from '@/components/ui/Button'
import { X } from 'lucide-react'
import { housekeepingApi } from '@/lib/api/housekeeping'

type ActionMode = 'idle' | 'confirm-reassign' | 'confirm-escalate' | 'confirm-acknowledge'

function PredictionRow({ prediction, canAssignRooms, onActionComplete }: {
  prediction: RoomPrediction
  canAssignRooms: boolean
  onActionComplete: () => void
}) {
  const { t } = useTranslation()
  const [mode, setMode] = useState<ActionMode>('idle')
  const [pending, setPending] = useState(false)
  const [resultNote, setResultNote] = useState<string | null>(null)
  // ...existing roomLabel / riskTone / layout unchanged above...

  const canAct = canAssignRooms && prediction.risk_level === 'HIGH'

  async function runAction(action: 'reassign' | 'escalate' | 'acknowledge') {
    setPending(true)
    setResultNote(null)
    try {
      if (action === 'reassign') {
        const res = await housekeepingApi.reassignAtRiskRoom(prediction.room_id)
        setResultNote(
          res.data.action === 'reassigned'
            ? t('housekeeping.predictionPanel.reassignedTo', { id: res.data.housekeeper_id })
            : t('housekeeping.predictionPanel.escalatedNoCapacity'),
        )
      } else if (action === 'escalate') {
        await housekeepingApi.escalateAtRiskRoom(prediction.room_id)
        setResultNote(t('housekeeping.predictionPanel.escalated'))
      } else {
        const res = await housekeepingApi.acknowledgeAtRiskRoom(prediction.room_id)
        setResultNote(
          res.data.action === 'already_acknowledged'
            ? t('housekeeping.predictionPanel.alreadyAcknowledged')
            : t('housekeeping.predictionPanel.acknowledged'),
        )
      }
      onActionComplete()
    } catch {
      setResultNote(t('housekeeping.predictionPanel.actionFailed'))
    } finally {
      setPending(false)
      setMode('idle')
    }
  }

  // Render: existing row content unchanged, plus (only when canAct):
  //   mode === 'idle'   -> three small IconButton/Button triggers that setMode('confirm-X')
  //   mode === 'confirm-X' -> inline "Confirm?" sub-row with a Confirm (loading={pending}, onClick=runAction) and Cancel (setMode('idle')) button, styled like LateCheckoutRow's confirm sub-rows (bg-surface-2/border-line, not the ready/alert-tinted variants those use for approve/deny, since these three actions aren't semantically approve/deny)
  //   resultNote, if set, renders as a small dismissable line below the row (clears on next fetchPredictions-driven re-render since resultNote is local state on a row keyed by room_id — acceptable, matches this panel's already-lightweight feedback style; no toast library exists in this codebase)
}
```
